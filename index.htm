Test 2: chemotaxis


%% paracond5
paracond.n = 1;
paracond.kappa = 1;

%% model scale
paracond.epsilon = 0.01;                             
paracond.gamma = 1;                                
paracond.a = paracond.gamma*paracond.epsilon^paracond.gamma;

%% chemotaxis
paracond.Q = @(t,x) (abs(2*x) < 1) .* (4/3 * (2/3 - abs(2*x).^2 + (abs(2*x).^3)/2)).*6 + ...
                    (abs(2*x) >= 1 && abs(2*x) < 2) .* (4/3 * (((2 - abs(2*x)).^3) / 6)).*6;  
paracond.Q_x = @(t,x) (abs(2*x) < 1) .* (x .* (-32/3 + 8 * abs(2*x))).*6 + ...
                    (abs(2*x) >= 1 && abs(2*x) < 2) .* (-4/3 * sign(x) .* (2 - abs(2*x)).^2).*6;
paracond.W = 1;                                                                   

%% fiber orientation
paracond.qx  = @(x) ones(size(x)) .* 1/2;         
paracond.e_v = @(v) sign(v);

paracond.qx_x = @(x) zeros(size(x));
